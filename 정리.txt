
insert into st_code_addr(ca_addr,ca_short_addr,ca_short_addr2,ca_ad1,ca_ad2,ca_ad3) 
SELECT concat(addr1,(case when addr2 <> '' then ' ' else '' end),addr2,(case when addr3 <> '' then ' ' else '' end),addr3,' ',addr4) as ca_addr
    ,concat(addr1,(case when addr2 <> '' then ' ' else '' end),addr2,(case when addr3 <> '' then ' ' else '' end),addr3,' ',addr4) as ca_short_addr
    ,concat(addr1,(case when addr2 <> '' then ' ' else '' end),addr2,(case when addr3 <> '' then ' ' else '' end),addr3) as ca_short_addr2
    ,addr1 as ca_ad1
    ,concat(addr2,(case when addr2 <> '' then ' ' else '' end),addr3) as ca_ad2
    ,addr4 as ca_ad3
FROM `temp_addr2` 
group by addr1,addr2,addr3,addr4;
